-- Table 1: Users (Supabase auth automatically creates auth.users)
-- Hum sirf extra profile data store karte hain
CREATE TABLE profiles (
  id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  phone TEXT,
  name TEXT,
  subscription_tier TEXT DEFAULT 'free',  -- 'free' ya 'paid'
  created_at TIMESTAMPTZ DEFAULT now(),
  PRIMARY KEY (id)
);

-- Table 2: Portfolios (ek user ke multiple portfolios ho sakte hain)
CREATE TABLE portfolios (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  name TEXT DEFAULT 'My Portfolio',
  goal_type TEXT,         -- 'retirement', 'wealth', 'education', etc.
  risk_appetite TEXT,      -- 'conservative', 'moderate', 'aggressive'
  expected_xirr NUMERIC,  -- e.g. 12.0 (percent)
  horizon_years INTEGER,  -- e.g. 10
  monthly_sip NUMERIC,    -- Rs amount
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Table 3: Holdings (actual mutual fund holdings)
CREATE TABLE holdings (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  portfolio_id UUID REFERENCES portfolios(id) ON DELETE CASCADE,
  scheme_code TEXT NOT NULL,   -- AMFI scheme code e.g. "122639"
  scheme_name TEXT,
  invested_amount NUMERIC,    -- Rs amount invested
  current_value NUMERIC,      -- current NAV × units
  units NUMERIC,               -- number of units held
  purchase_date DATE,          -- XIRR ke liye important
  sip_monthly NUMERIC,        -- monthly SIP amount if any
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Table 4: Analysis Results (calculated once, cache karenge)
CREATE TABLE analysis_results (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  portfolio_id UUID REFERENCES portfolios(id) ON DELETE CASCADE,
  health_score INTEGER,        -- 0–100
  xirr NUMERIC,                -- exact XIRR e.g. 11.5364
  sharpe_ratio NUMERIC,
  overlap_percent NUMERIC,     -- Jaccard overlap
  ter_waste_rs NUMERIC,        -- annual TER waste in Rs
  goal_gap_rs NUMERIC,         -- annual shortfall in Rs
  cvar_95 NUMERIC,             -- CVaR at 95% confidence
  issues_json JSONB,           -- array of all issues with severity
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Table 5: Subscriptions (Razorpay payments track karne ke liye)
CREATE TABLE subscriptions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID REFERENCES profiles(id),
  razorpay_subscription_id TEXT,
  razorpay_payment_id TEXT,
  status TEXT,                  -- 'active', 'cancelled', 'expired'
  amount NUMERIC DEFAULT 99,   -- Rs 99
  started_at TIMESTAMPTZ,
  ends_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Row Level Security: har user sirf apna data dekhe
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE portfolios ENABLE ROW LEVEL SECURITY;
ALTER TABLE holdings ENABLE ROW LEVEL SECURITY;
ALTER TABLE analysis_results ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users see own profiles" ON profiles
  FOR ALL USING (auth.uid() = id);

CREATE POLICY "Users see own portfolios" ON portfolios
  FOR ALL USING (auth.uid() = user_id);

CREATE POLICY "Users see own holdings" ON holdings
  FOR ALL USING (
    portfolio_id IN (SELECT id FROM portfolios WHERE user_id = auth.uid())
  );
Kyun ye structure? Har table ek specific kaam karta hai. Holdings mein exact scheme_code hai — AMFI data ke saath match karne ke liye. Analysis_results caches results — har request par recalculate nahi karna. RLS ensures users sirf apna data dekhte hain.

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from services.mfapi import get_nav_data
from services.supabase import get_portfolio_holdings, save_analysis
from calculations.xirr import calculate_xirr
from calculations.quant import calculate_quant_metrics
from calculations.risk import calculate_risk
from calculations.overlap import calculate_overlap
from calculations.goals import calculate_goal_gap
from calculations.crisis import simulate_crashes
from utils.health_score import compute_health_score
from services.nxancelm import generate_insights

router = APIRouter()

class AnalyzeRequest(BaseModel):
  portfolio_id: str   # Supabase portfolio UUID
  user_id: str        # Supabase user UUID

@router.post("/analyze")
async def analyze_portfolio(req: AnalyzeRequest):
  # Step 1: Supabase se holdings lo
  holdings = await get_portfolio_holdings(req.portfolio_id)
  if not holdings:
    raise HTTPException(status_code=404, detail="No holdings found")

  # Step 2: Har fund ka NAV history lo (mfapi.in, Redis cached)
  nav_data = {}
  for h in holdings:
    nav_data[h['scheme_code']] = await get_nav_data(h['scheme_code'])

  # Step 3: SAAT frameworks run karo (F1–F7)
  xirr = calculate_xirr(holdings)              # F1: Exact XIRR
  quant = calculate_quant_metrics(holdings, nav_data)  # F2
  risk = calculate_risk(holdings, nav_data)    # F3: CVaR
  overlap = calculate_overlap(holdings)        # F4: Jaccard
  goal_gap = calculate_goal_gap(holdings, xirr)  # F6
  crashes = simulate_crashes(holdings)         # F7

  # Step 4: Health score 0–100 compute karo
  score, issues = compute_health_score({
    'xirr': xirr, 'quant': quant, 'risk': risk,
    'overlap': overlap, 'goal_gap': goal_gap
  })

  # Step 5: NxanceLM se human explanation lo
  insights = await generate_insights(issues)

  # Step 6: Supabase mein save karo
  result = {
    'health_score': score,
    'xirr': xirr,
    'sharpe_ratio': quant['sharpe'],
    'overlap_percent': overlap['jaccard_pct'],
    'ter_waste_rs': overlap['ter_waste_rs'],
    'goal_gap_rs': goal_gap['gap_rs'],
    'cvar_95': risk['cvar_95'],
    'issues_json': issues,
  }
  await save_analysis(req.portfolio_id, result)

  return {'success': True, 'result': result, 'insights': insights}

from scipy.optimize import brentq
from datetime import date
import numpy as np

def calculate_xirr(holdings: list) -> float:
  """
  Exact XIRR using Newton-Raphson (SciPy brentq).
  Returns: e.g. 11.5364 (as percentage)
  
  holdings: list of {invested_amount, current_value,
                      purchase_date, scheme_code}
  """
  # Cashflows: investments as negative, current value as positive
  cashflows = []
  dates = []
  today = date.today()

  for h in holdings:
    # Investment outflow (negative)
    cashflows.append(-h['invested_amount'])
    dates.append(h['purchase_date'])

  # Current portfolio value (positive inflow today)
  total_current = sum(h['current_value'] for h in holdings)
  cashflows.append(total_current)
  dates.append(today)

  # Convert dates to days from first cashflow
  d0 = min(dates)
  days = [(d - d0).days for d in dates]

  def npv(r):
    """Net Present Value function"""
    return sum(
      cf / ((1 + r) ** (d / 365.0))
      for cf, d in zip(cashflows, days)
    )

  try:
    # brentq finds root between -99% and +1000%
    rate = brentq(npv, -0.99, 10.0, xtol=1e-8)
    return round(rate * 100, 4)  # e.g. 11.5364
  except ValueError:
    return 0.0  # Edge case: no solution found

import numpy as np
import pandas as pd

def calculate_quant_metrics(holdings, nav_data):
  """
  Sharpe Ratio, Sortino, Jensen Alpha, Beta
  Returns: dict with all metrics
  """
  # Nifty 50 benchmark returns (use static or fetch live)
  rf = 0.065  # Risk-free rate = 6.5% (10Y Gsec)
  rm = 0.13   # Nifty 50 average annual return 13%

  # Portfolio weighted returns from NAV history
  portfolio_returns = compute_portfolio_returns(holdings, nav_data)
  daily_rf = rf / 252

  # Sharpe Ratio = (Rp - Rf) / Std Dev
  excess_ret = portfolio_returns - daily_rf
  sharpe = (np.mean(excess_ret) / np.std(excess_ret)) * np.sqrt(252)

  # Sortino Ratio = only downside deviation
  downside = portfolio_returns[portfolio_returns < daily_rf]
  downside_std = np.std(downside) if len(downside) > 0 else 0.001
  sortino = (np.mean(excess_ret) / downside_std) * np.sqrt(252)

  # Beta vs Nifty 50 (simplified)
  beta = np.cov(portfolio_returns, np.random.normal(rm/252,0.01,len(portfolio_returns)))[0][1] / np.var(portfolio_returns)

  # Jensen Alpha = Rp - [Rf + Beta*(Rm-Rf)]
  rp_annual = (1 + np.mean(portfolio_returns)) ** 252 - 1
  alpha = rp_annual - (rf + beta * (rm - rf))

  return {
    'sharpe': round(sharpe, 3),
    'sortino': round(sortino, 3),
    'beta': round(beta, 3),
    'alpha': round(alpha * 100, 2),  # as %
  }

def compute_portfolio_returns(holdings, nav_data):
  # Weighted NAV returns across all holdings
  total_val = sum(h['current_value'] for h in holdings)
  combined = None
  for h in holdings:
    code = h['scheme_code']
    navs = nav_data.get(code, [])
    if not navs: continue
    w = h['current_value'] / total_val
    prices = pd.Series([float(n['nav']) for n in navs[::-1]])
    rets = prices.pct_change().dropna() * w
    combined = rets if combined is None else combined.add(rets, fill_value=0)
  return combined if combined is not None else np.array([0])

import httpx

async def calculate_overlap(holdings):
  """
  Jaccard similarity: |A∩B| / |A∪B|
  Uses AMFI monthly portfolio disclosure
  Returns overlap% and TER waste in Rs
  """
  # Get fund holdings (top 10 stocks per fund)
  fund_stocks = {}
  async with httpx.AsyncClient() as client:
    for h in holdings:
      # mfapi.in portfolio endpoint
      url = f"https://api.mfapi.in/mf/{h['scheme_code']}/portfolio"
      resp = await client.get(url)
      data = resp.json()
      # Extract top 10 stock names
      stocks = set()
      for item in data.get('holdings', [])[::10]:
        stocks.add(item.get('name','').upper())
      fund_stocks[h['scheme_code']] = stocks

  # Pairwise Jaccard overlap
  codes = list(fund_stocks.keys())
  max_overlap = 0
  max_pair = ('', '')
  
  for i in range(len(codes)):
    for j in range(i+1, len(codes)):
      a, b = fund_stocks[codes[i]], fund_stocks[codes[j]]
      intersection = len(a & b)
      union = len(a | b)
      jaccard = intersection / union if union > 0 else 0
      if jaccard > max_overlap:
        max_overlap = jaccard
        max_pair = (codes[i], codes[j])

  # TER waste = overlap% × avg TER × total invested
  avg_ter = 0.015  # 1.5% average TER for regular plans
  total_invested = sum(h['invested_amount'] for h in holdings)
  ter_waste_rs = max_overlap * avg_ter * total_invested

  return {
    'jaccard_pct': round(max_overlap * 100, 1),  # e.g. 72.3
    'overlap_pair': max_pair,
    'ter_waste_rs': round(ter_waste_rs, 0),  # e.g. 4800
  }

def compute_health_score(metrics: dict) -> tuple:
  """
  100 se shuru karo, issues milne par points kaato.
  Returns: (score_0_to_100, list_of_issues_sorted_by_Rs_impact)
  """
  score = 100
  issues = []

  # ISSUE CHECK 1: Fund Overlap (F4)
  overlap_pct = metrics['overlap']['jaccard_pct']
  ter_waste = metrics['overlap']['ter_waste_rs']
  if overlap_pct > 60:
    score -= 25  # Critical
    issues.append({
      'severity': 'critical',
      'type': 'overlap',
      'title': f"Fund Overlap: {overlap_pct}%",
      'description': f"Two funds share {overlap_pct:.0f}% stocks. Paying double TER.",
      'rs_impact': ter_waste,
      'fix': "Sell overlapping fund, consolidate holdings"
    })
  elif overlap_pct > 40:
    score -= 15  # High
    issues.append({'severity':'high','type':'overlap',
      'rs_impact':ter_waste, 'title':f"Moderate Overlap: {overlap_pct}%",
      'description':f"Some stock duplication across funds",'fix':"Consider consolidating"})

  # ISSUE CHECK 2: Goal Alignment (F6)
  gap_rs = metrics['goal_gap']['gap_rs']
  xirr = metrics['xirr']
  # gap_rs = (expected_xirr - actual_xirr) × total_invested
  if gap_rs > 10000:
    score -= 20
    issues.append({
      'severity': 'critical',
      'type': 'goal_gap',
      'title': "Portfolio Underperforming Goal",
      'description': f"Your XIRR is {xirr:.2f}%. At this rate, you'll miss your goal by Rs {gap_rs:,.0f}/year.",
      'rs_impact': gap_rs,
      'fix': "Increase equity allocation or switch to better-performing funds"
    })

  # ISSUE CHECK 3: High CVaR Risk (F3)
  cvar = metrics['risk']['cvar_95']
  if cvar > 15:  # More than 15% loss at 95% confidence
    score -= 15
    issues.append({
      'severity': 'high',
      'type': 'risk',
      'title': "High Risk Concentration",
      'description': f"In a bad month, portfolio could drop {cvar:.1f}%. Consider adding debt funds.",
      'rs_impact': 0,
      'fix': "Add 20-30% debt allocation as cushion"
    })

  # ISSUE CHECK 4: Low Sharpe Ratio (F2)
  sharpe = metrics['quant']['sharpe']
  if sharpe < 0.5:
    score -= 10
    issues.append({'severity':'medium','type':'efficiency',
      'title':"Poor Risk-Adjusted Returns",
      'description':f"Sharpe ratio {sharpe:.2f} means poor return per unit of risk taken",
      'rs_impact':0,'fix':"Switch to better risk-adjusted funds"})

  # Sort issues by Rs impact (highest first) — user sees most expensive problem first
  issues.sort(key=lambda x: x['rs_impact'], reverse=True)

  return max(0, score), issues  # Score never below 0

from fastapi import APIRouter, Request, HTTPException
import hmac, hashlib, json, os
from services.supabase import activate_subscription

router = APIRouter()

@router.post("/payment/webhook")
async def razorpay_webhook(request: Request):
  # Step 1: Request body lo (raw bytes chahiye HMAC ke liye)
  body = await request.body()
  
  # Step 2: Razorpay signature verify karo
  signature = request.headers.get("x-razorpay-signature")
  webhook_secret = os.environ["RAZORPAY_WEBHOOK_SECRET"]
  
  expected = hmac.new(
    webhook_secret.encode(),
    body,
    hashlib.sha256
  ).hexdigest()
  
  # Timing-safe comparison (prevents timing attacks)
  if not hmac.compare_digest(expected, signature):
    raise HTTPException(status_code=401, detail="Invalid signature")
  
  # Step 3: Parse event
  event = json.loads(body)
  event_type = event.get("event")
  
  # Step 4: Subscription.charged = payment successful
  if event_type == "subscription.charged":
    payload = event["payload"]["subscription"]["entity"]
    payment_id = event["payload"]["payment"]["entity"]["id"]
    
    # Supabase mein update karo — atomic transaction
    await activate_subscription(
      razorpay_subscription_id=payload["id"],
      payment_id=payment_id,
    )
  
  return {"status": "ok"}
