-- Table 1: Users/Profiles
-- auth.users automatically created by Supabase when OTP login hota hai
-- Hum sirf extra info store karte hain
CREATE TABLE public.profiles (
  id          UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  phone       TEXT,
  email       TEXT,
  name        TEXT,
  subscription_tier TEXT DEFAULT 'free',  -- 'free' ya 'founding' (Rs99/month)
  created_at  TIMESTAMPTZ DEFAULT now(),
  updated_at  TIMESTAMPTZ DEFAULT now(),
  PRIMARY KEY (id)
);

-- Table 2: Portfolios
-- User ke paas ek hi portfolio hoga Phase 1 mein
-- Phase 2 mein multiple portfolios ka plan hai
CREATE TABLE public.portfolios (
  id           UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id      UUID REFERENCES profiles(id) ON DELETE CASCADE,
  name         TEXT DEFAULT 'My Portfolio',
  -- Goal questions se aata hai (5 MCQ answers)
  goal_type    TEXT,   -- 'retirement' | 'wealth' | 'house' | 'education' | 'emergency'
  risk_level   TEXT,   -- 'conservative' | 'moderate' | 'aggressive' | 'ultra'
  expected_xirr NUMERIC,  -- e.g. 12.0 (percent)
  horizon_years INTEGER,  -- e.g. 10 years
  monthly_sip  NUMERIC,  -- Rs amount user can invest per month
  created_at   TIMESTAMPTZ DEFAULT now()
);

-- Table 3: Holdings (actual mutual fund holdings)
-- Financial data = AES-256-GCM encrypt karte hain (DPDPA compliance)
CREATE TABLE public.holdings (
  id               UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  portfolio_id     UUID REFERENCES portfolios(id) ON DELETE CASCADE,
  scheme_code      TEXT NOT NULL,   -- AMFI scheme code e.g. "119597" (HDFC Flexicap Direct)
  scheme_name      TEXT,            -- Full fund name
  invested_amount  NUMERIC,         -- Total Rs invested (encrypted later)
  current_value    NUMERIC,         -- Current NAV x units
  units            NUMERIC,         -- Number of units held
  purchase_date    DATE,            -- First investment date (critical for XIRR)
  sip_monthly      NUMERIC,         -- Monthly SIP amount (0 if lump sum)
  plan_type        TEXT DEFAULT 'direct',  -- 'direct' ya 'regular'
  created_at       TIMESTAMPTZ DEFAULT now()
);

-- Table 4: Analysis Results
-- Har analysis ek row = same portfolio multiple times analyse ho sakta hai
CREATE TABLE public.analysis_results (
  id               UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  portfolio_id     UUID REFERENCES portfolios(id) ON DELETE CASCADE,
  health_score     INTEGER,     -- 0 to 100
  xirr             NUMERIC,     -- Exact XIRR e.g. 11.5364
  sharpe_ratio     NUMERIC,     -- e.g. 0.42
  sortino_ratio    NUMERIC,
  jensen_alpha     NUMERIC,     -- vs Nifty 50 benchmark
  beta             NUMERIC,
  max_drawdown     NUMERIC,
  overlap_pct      NUMERIC,     -- Jaccard overlap % between any 2 funds
  ter_waste_rs     NUMERIC,     -- Annual TER waste in Rs
  goal_gap_rs      NUMERIC,     -- Annual Rs shortfall from goal
  cvar_95          NUMERIC,     -- Conditional Value at Risk (95%)
  issues_json      JSONB,       -- Array of all issues {type, severity, rs_impact, fix}
  insights_text    TEXT,        -- NxanceLM generated Hinglish explanation
  created_at       TIMESTAMPTZ DEFAULT now()
);

-- Table 5: Subscriptions (Razorpay payments track karne ke liye)
CREATE TABLE public.subscriptions (
  id                     UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id                UUID REFERENCES profiles(id),
  razorpay_order_id      TEXT UNIQUE,  -- Prevent duplicate processing
  razorpay_payment_id    TEXT,
  amount                 NUMERIC DEFAULT 99,  -- Rs 99
  status                 TEXT DEFAULT 'pending',  -- 'pending' | 'paid' | 'failed'
  paid_at                TIMESTAMPTZ,
  expires_at             TIMESTAMPTZ,         -- 30 days after payment
  created_at             TIMESTAMPTZ DEFAULT now()
);

-- ═══════════════════════════════════════════════════════
-- ROW LEVEL SECURITY (RLS) — CRITICAL for data privacy
-- Har user sirf apna data dekhe, kisi aur ka nahi
-- ═══════════════════════════════════════════════════════
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE portfolios ENABLE ROW LEVEL SECURITY;
ALTER TABLE holdings ENABLE ROW LEVEL SECURITY;
ALTER TABLE analysis_results ENABLE ROW LEVEL SECURITY;
ALTER TABLE subscriptions ENABLE ROW LEVEL SECURITY;

-- RLS Policies — simple rule: auth.uid() (logged in user ID) = user_id
CREATE POLICY "users own profiles" ON profiles
  FOR ALL USING (auth.uid() = id);

CREATE POLICY "users own portfolios" ON portfolios
  FOR ALL USING (auth.uid() = user_id);

CREATE POLICY "users own holdings" ON holdings
  FOR ALL USING (
    portfolio_id IN (
      SELECT id FROM portfolios WHERE user_id = auth.uid()
    )
  );

CREATE POLICY "users own analysis" ON analysis_results
  FOR ALL USING (
    portfolio_id IN (
      SELECT id FROM portfolios WHERE user_id = auth.uid()
    )
  );

CREATE POLICY "users own subscriptions" ON subscriptions
  FOR ALL USING (auth.uid() = user_id);

-- Auto-create profile when user signs up (OTP phone login ke baad)
-- auth.users mein new row → profiles mein automatically insert
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, phone, email)
  VALUES (NEW.id, NEW.phone, NEW.email);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE handle_new_user();

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from routers import analyze, payment, health
import os

app = FastAPI(
    title="Nxance API",
    description="Portfolio health analytics",
    version="1.0.0"
)

# CORS — Lovable frontend aur nxance.in ko allow karo
app.add_middleware(
    CORSMiddleware,
    allow_origins=[
        "https://nxance.in",
        "https://*.lovable.app",
        "http://localhost:5173",
    ],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Routers register karo
app.include_router(analyze.router, prefix="/api")
app.include_router(payment.router, prefix="/api")
app.include_router(health.router, prefix="/api")

@app.get("/")
async def root():
    return {"message": "Nxance API v1.0"}

F1: XIRR — Newton-Raphson exact
F1 — XIRR formula
NPV(r) = Σ cashflow_i / (1+r)^(days_i/365) = 0
SciPy brentq root-finder solves for r. Result: 11.5364% not "roughly 11–12%". Handles every SIP date separately — most accurate return metric for Indian investors with irregular SIPs.
PYTHONcalculations/xirr.py
from scipy.optimize import brentq
from datetime import date
import numpy as np

def calculate_xirr(holdings: list) -> float:
    """Exact XIRR using Newton-Raphson (SciPy brentq).
    holdings: [{invested_amount, current_value, 
               purchase_date, scheme_code}]
    Returns: e.g. 11.5364 (as percent)"""
    cashflows, dates = [], []
    today = date.today()
    
    for h in holdings:
        cashflows.append(-h['invested_amount'])
        dates.append(h['purchase_date'])
    
    total = sum(h['current_value'] for h in holdings)
    cashflows.append(total)
    dates.append(today)
    
    d0 = min(dates)
    days = [(d - d0).days for d in dates]
    
    def npv(r):
        return sum(
            cf / ((1+r)**(d/365.0))
            for cf, d in zip(cashflows, days)
        )
    try:
        rate = brentq(npv, -0.99, 10.0, xtol=1e-8)
        return round(rate * 100, 4)
    except:
        return 0.0
F4: Jaccard Overlap — set theory
F4 — Jaccard similarity
Jaccard = |A ∩ B| / |A ∪ B|
A = Fund 1 top-10 stocks, B = Fund 2 top-10 stocks. 72% Jaccard = 72% of stocks same = paying double TER for duplicate exposure. TER waste = Jaccard% × avg_TER × corpus = exact Rs/year.
PYTHONcalculations/overlap.py
import httpx

async def calculate_overlap(holdings):
    """Jaccard overlap |A∩B|/|A∪B| on AMFI monthly holdings.
    Returns: {jaccard_pct, ter_waste_rs, worst_pair}"""
    fund_stocks = {}
    
    async with httpx.AsyncClient() as c:
        for h in holdings:
            # mfapi.in has monthly portfolio disclosure
            url = f"https://api.mfapi.in/mf/{h['scheme_code']}/portfolio"
            resp = await c.get(url, timeout=5)
            data = resp.json()
            stocks = set()
            for item in data.get('holdings', [])[::10]:
                stocks.add(item.get('name','').upper().strip())
            fund_stocks[h['scheme_code']] = stocks
    
    codes = list(fund_stocks.keys())
    max_j, best_pair = 0, ('','')
    
    for i in range(len(codes)):
        for j in range(i+1, len(codes)):
            a = fund_stocks[codes[i]]
            b = fund_stocks[codes[j]]
            union = len(a | b)
            j_val = len(a & b) / union if union > 0 else 0
            if j_val > max_j:
                max_j, best_pair = j_val, (codes[i], codes[j])
    
    avg_ter = 0.015  # 1.5% avg TER regular plans
    corpus = sum(h['invested_amount'] for h in holdings)
    waste = max_j * avg_ter * corpus
    
    return {
        'jaccard_pct': round(max_j * 100, 1),
        'ter_waste_rs': round(waste, 0),
        'worst_pair': best_pair
    }
F3: CVaR + F6: Goal Gap + Health Score
F3 — CVaR parametric
CVaR = μ - σ × φ(Φ⁻¹(α)) / (1-α)
95% confidence mein worst case loss. α=0.05. σ = portfolio volatility. If CVaR > 15% = High risk issue. Applied to 2008 crash (−52%) and 2020 COVID (−38%) scenarios.
F6 — Goal alignment gap
gap_rs = (expected_XIRR − actual_XIRR) × corpus
If expected 12%, actual 9.2%, corpus Rs 5L: gap = 2.8% × 5,00,000 = Rs 14,000/year shortfall. This exact Rs number is shown to user.
Health Score formula
Score = 100 − Σ severity_deductions
Critical issue: −25pts. High: −15pts. Medium: −8pts. Low: −3pts. Issues sorted by Rs impact (highest first). Score never below 0. Green ≥70, Yellow 40–69, Red <40.

def compute_health_score(metrics):
    score = 100; issues = []
    
    # F4: Overlap check
    olap = metrics['overlap']['jaccard_pct']
    waste = metrics['overlap']['ter_waste_rs']
    if olap > 60: score -= 25; sev='critical'
    elif olap > 40: score -= 15; sev='high'
    elif olap > 20: score -= 8; sev='medium'
    else: sev = None
    if sev: issues.append({
        'type':'overlap','severity':sev,
        'rs_impact':waste,
        'title':f"{olap}% fund overlap",
        'fix':"Sell higher-TER fund"})
    
    # F6: Goal gap check
    gap = metrics['goal_gap']['gap_rs']
    if gap > 10000: score -= 20
    elif gap > 3000: score -= 10
    
    # Sort by Rs impact — biggest problem first
    issues.sort(key=lambda x: x['rs_impact'], 
                  reverse=True)
    return max(0, score), issues

r = redis.from_url(os.environ.get("REDIS_URL", "redis://localhost"))

async def get_nav_history(scheme_code: str) -> list:
    """mfapi.in se NAV history fetch karo.
    Redis cache check first (TTL 24 hours).
    Free API — no key needed."""
    cache_key = f"nav:{scheme_code}"
    
    # Cache check karo
    cached = r.get(cache_key)
    if cached:
        return json.loads(cached)
    
    # mfapi.in se fetch karo
    url = f"https://api.mfapi.in/mf/{scheme_code}"
    async with httpx.AsyncClient(timeout=10) as client:
        resp = await client.get(url)
        data = resp.json()
    
    nav_list = data.get('data', [])
    # Format: [{date: "05-06-2025", nav: "45.23"}]
    
    # Cache mein store karo (24 hours)
    r.setex(cache_key, 86400, json.dumps(nav_list))
    return nav_list

async def get_fund_holdings(scheme_code: str) -> list:
    """Fund's monthly stock holdings fetch karo.
    AMFI monthly disclosure — for Jaccard overlap."""
    cache_key = f"holdings:{scheme_code}"
    
    cached = r.get(cache_key)
    if cached:
        return json.loads(cached)
    
    url = f"https://api.mfapi.in/mf/{scheme_code}/portfolio"

        data = resp.json()
    
    holdings = data.get('holdings', [])

from transformers import AutoModelForCausalLM, AutoTokenizer
from peft import PeftModel
import torch, asyncio, os, redis, json, hashlib

# Redis client
r = redis.from_url(os.environ.get("REDIS_URL"))

# Model global load (ek baar load — har request pe nahi)
# AWS g4dn.xlarge T4 GPU par run karega
_model = None
_tokenizer = None
_lock = asyncio.Lock()  # Ek GPU — ek time par ek request

def load_model():
    global _model, _tokenizer
    if _model is None:
        base = AutoModelForCausalLM.from_pretrained(
            "TinyLlama/TinyLlama-1.1B-Chat-v1.0",
            torch_dtype=torch.float16,
            device_map="auto"
        )
        _model = PeftModel.from_pretrained(
            base, os.environ["NXANCELM_ADAPTER_PATH"]
        )
        _tokenizer = AutoTokenizer.from_pretrained(
            os.environ["NXANCELM_ADAPTER_PATH"]
        )

def get_template(issues: list) -> str:
    """40% of cases: pre-written template (fastest, Rs 0, 0ms)."""
    if not issues:
        return "Portfolio bilkul theek hai! Koi bada issue nahi mila. SIP continue rakho."
    top = issues[0]
    if top['type'] == 'overlap' and top['severity'] == 'critical':
        return f"Fund overlap critical issue hai — Rs{top['rs_impact']:,.0f}/year waste. Fix: overlapping fund sell karo."
    return None  # Template nahi mila — model use karo

async def generate_insights(issues: list, tier: str = "free") -> str:
    """Smart router:
    1. Template (40% cases) — Rs 0, 0ms
    2. Redis cache (30% cases) — Rs 0, <1ms  
    3. TinyLlama (30% cases) — Rs 0, 2-4s
    """
    max_tokens = 80 if tier == "free" else 250
    
    # Route 1: Template check
    template = get_template(issues)
    if template:
        return template
    
    # Route 2: Redis cache (similar portfolios)
    issue_key = hashlib.md5(json.dumps(issues, sort_keys=True).encode()).hexdigest()[:12]
    cache_key = f"insight:{issue_key}:{tier}"
    cached = r.get(cache_key)
    if cached:
        return cached.decode()
    
    # Route 3: TinyLlama inference
    prompt = build_prompt(issues, max_tokens)
    
    async with _lock:  # Ek time par ek request GPU pe
        load_model()
        inputs = _tokenizer(prompt, return_tensors="pt").to("cuda")
        with torch.no_grad():
            outputs = _model.generate(
                **inputs,

@router.post("/payment/webhook")
async def razorpay_webhook(request: Request):
    """HMAC-SHA256 se verify karo — fake webhooks reject karo."""
    
    # Step 1: Raw bytes lo (hmac ke liye raw body chahiye)
    body = await request.body()
    signature = request.headers.get("x-razorpay-signature", "")
    
    # Step 2: Razorpay signature verify karo
    secret = os.environ["RAZORPAY_WEBHOOK_SECRET"]
    expected_sig = hmac.new(
        secret.encode(),
        body,
        hashlib.sha256
    ).hexdigest()
    
    # Timing-safe comparison (prevents timing attacks)
    if not hmac.compare_digest(expected_sig, signature):
        raise HTTPException(status_code=401, detail="Invalid signature")
    
    # Step 3: Event parse karo
    event = json.loads(body)
    event_type = event.get("event")
    
    # Step 4: Payment captured = subscription activate karo
    if event_type == "payment.captured":
        payment = event["payload"]["payment"]["entity"]
        order_id = payment["order_id"]
        payment_id = payment["id"]
        user_id = payment["notes"].get("user_id")
        
        if user_id:
            await activate_subscription(order_id, payment_id, user_id)
    
    return {"status": "ok"}

async def activate_subscription(order_id, payment_id, user_id):
    """Atomic transaction: payment insert + tier upgrade.
    Agar koi bhi fail hota hai — dono rollback. No partial state."""
    from datetime import datetime, timedelta
    
    # Duplicate check — same order_id dobara process mat karo
    existing = supabase.table("subscriptions")\
        .select("id").eq("razorpay_order_id", order_id).execute()
    
    if existing.data:
        return  # Already processed
    
    expires = datetime.utcnow() + timedelta(days=31)
    
    # INSERT subscription record
    supabase.table("subscriptions").insert({
        "user_id": user_id,
        "razorpay_order_id": order_id,
        "razorpay_payment_id": payment_id,
        "status": "paid",
        "paid_at": datetime.utcnow().isoformat(),
        "expires_at": expires.isoformat(),
        "amount": 99
    }).execute()
    
    # UPDATE user tier — founding unlock
    supabase.table("profiles")\
        .update({"subscription_tier": "founding"})\
        .eq("id", user_id)\
        .execute()

HMAC-SHA256 kyun zaroori hai — security explained
🔐
HMAC-SHA256 Webhook Verification
Security Critical
Problem
Koi bhi /api/payment/webhook pe fake POST request bhej sakta hai pretending payment hua. Bina verification: free subscription mil jaaye bina pay kiye.
Solution
Razorpay apne secret key se request body sign karta hai. Server verify karta hai — secret key sirf Razorpay aur tumhare server ko pata hai.
How it works
HMAC(secret_key, request_body) = signature. Ye signature header mein aata hai. Server same calculation karta hai — match hone par valid.
hmac.compare_digest
Simple == comparison: timing attack possible (attacker can guess byte by byte). compare_digest: always constant time — timing attack impossible.
UNIQUE constraint
order_id UNIQUE constraint in subscriptions table prevents same webhook processed twice even if Razorpay retries.
